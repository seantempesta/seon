(ns scratch.tasks-v01
  "Experiment v0.1: authored tasks carry only id, intent, subjects, target schemas and done-when;
  deliverables, order, changed keys, context and state are DERIVED. Throwaway."
  (:require [clojure.string :as str]
            [datahike.api :as d]
            [edamame.core :as e]
            [malli.core :as m]
            [malli.error :as me]
            [malli.registry :as mr]))

(def authored-schema
  "What an author writes. Everything else is derived; overrides are allowed but rare."
  [:map {:closed true}
   [:seon.task/id :qualified-keyword]
   [:seon.task/intent [:string {:min 20}]]
   [:seon.task/subjects [:set {:min 1} :symbol]]
   [:seon.task/schemas {:optional true} [:map-of :qualified-keyword :any]]
   [:seon.task/done-when {:optional true} [:vector {:min 1} :any]]
   [:seon/must-read {:optional true} [:set :qualified-symbol]]
   [:seon.task/evidence {:optional true} [:set :string]]
   [:seon.task/entities {:optional true} [:set [:tuple :qualified-keyword :any]]]  ; stored data it is about
   [:seon.task/changes {:optional true} [:set :symbol]]   ; narrows what may change; #{} = data-only task
   [:seon.task/recurs {:optional true} [:map [:every [:enum :hour :day :week]]]]
   [:seon.task/delivers {:optional true} [:set :qualified-symbol]]  ; override only
   [:seon.task/needs {:optional true} [:set :qualified-keyword]]])  ; override only

(defn read-dir
  "Every task in every EDN file under `dir`, read as Clojure data without evaluation."
  [dir]
  (vec (mapcat #(e/parse-string (slurp %) {:quote true :deref false :fn false :regex false})
               (filter #(str/ends-with? (str %) ".edn") (file-seq (java.io.File. dir))))))

(defn exists?
  "A program row, a loaded Var, or a namespace row."
  [db s]
  (boolean (or (d/q '[:find ?e . :in $ ?s :where [?e :seon.fn/sym ?s]] db s)
               (d/q '[:find ?e . :in $ ?s :where [?e :seon.test/sym ?s]] db s)
               (d/q '[:find ?e . :in $ ?s :where [?e :seon.ns/name ?s]] db s)
               (try (find-var s) (catch Exception _ nil)))))

(defn referenced
  "Every qualified symbol a task's done-when forms mention (called or quoted)."
  [task]
  (set (filter qualified-symbol? (tree-seq coll? seq (:seon.task/done-when task)))))

(defn derive-task
  "Authored task + database value -> the full task. Pure."
  [db task]
  (let [missing (set (remove #(exists? db %) (referenced task)))
        scope (or (:seon.task/changes task) (:seon.task/subjects task))   ; what may change
        fn-subjects (set (filter qualified-symbol? scope))
        ;; a task delivers a missing symbol only when the symbol's namespace is in what it may change
        subject-nss (set (map #(if (qualified-symbol? %) (symbol (namespace %)) %) scope))
        owned-missing (set (filter #(subject-nss (symbol (namespace %))) missing))]
    (cond-> (assoc task
                   :seon.task/keys (set (keys (:seon.task/schemas task)))
                   :seon.task/context (if (seq (:seon.task/done-when task)) 'seon.task/implement 'seon.task/triage)
                   :seon.task/state (if (seq (:seon.task/done-when task)) :specified :raised)
                   ::missing missing)
      (not (:seon.task/delivers task))
      (assoc :seon.task/delivers (into fn-subjects owned-missing)))))

(defn derive-needs
  "Task B needs task A when B's done-when references something A delivers (and A is not B)."
  [tasks]
  (let [deliverers (reduce (fn [m [s id]] (update m s (fnil conj (sorted-set)) id)) {}
                            (for [t tasks s (:seon.task/delivers t)] [s (:seon.task/id t)]))]
    (mapv (fn [t]
            (let [derived (set (mapcat #(disj (get deliverers % #{}) (:seon.task/id t)) (referenced t)))]
              (assoc t :seon.task/needs (into derived (:seon.task/needs t)))))
          tasks)))

(defn findings
  "Authored shape, then the v0 rules over the derived tasks."
  [db authored derived]
  (let [by-id (into {} (map (juxt :seon.task/id identity)) derived)
        delivered-by-needs (fn [t] (set (mapcat #(:seon.task/delivers (by-id %)) (:seon.task/needs t))))]
    (vec (concat
          (for [[s owners] (group-by first (for [t derived s (:seon.task/delivers t)
                                                 :when (not (exists? db s))] [s (:seon.task/id t)]))
                :when (> (count owners) 1)]
            {:task (vec (sort (map second owners))) :rule :deliverable-claimed-twice :sym s})
          (for [a authored :let [ex (m/explain authored-schema a)] :when ex]
            {:task (:seon.task/id a) :rule :shape :why (me/humanize ex)})
          (for [t derived
                f (concat
                   (for [s (:seon.task/subjects t)
                         :when (not (or (exists? db s)
                                        (and (simple-symbol? s) (some #(= (str s) (namespace %)) (referenced t)))))]
                     {:rule :subject-missing :sym s})
                   (for [s (:seon/must-read t) :when (not (exists? db s))] {:rule :must-read-missing :sym s})
                   (for [s (::missing t) :when (not (or ((:seon.task/delivers t) s) ((delivered-by-needs t) s)))]
                     {:rule :references-unknown :sym s})
                   (for [p (:seon.task/evidence t) :when (not (.exists (java.io.File. (str "/Users/sean/src/seon/" p))))]
                     {:rule :evidence-missing :path p}))]
            (assoc f :task (:seon.task/id t)))))))

(defn plan
  "Tasks in derived dependency order with their derived state."
  [derived]
  (let [by-id (into {} (map (juxt :seon.task/id identity)) derived)]
    (loop [out [] done #{} left (set (keys by-id))]
      (if (empty? left) out
          (let [ready (sort (filter #(every? done (:seon.task/needs (by-id %))) left))]
            (if (empty? ready)
              (conj out {:cycle-or-unknown-prerequisite (vec (sort left))})
              (recur (into out (for [id ready :let [t (by-id id)]]
                                 {:id id :state (:seon.task/state t) :needs (vec (sort (:seon.task/needs t)))
                                  :delivers (vec (sort (:seon.task/delivers t)))}))
                     (into done ready) (reduce disj left ready))))))))

(defn branch-evaluator
  "Evaluate source strings in the SCI context of `branch` on `cluster-name` (writes stay on the branch)."
  [cluster-name branch]
  (let [instance (get @@(ns-resolve 'seon.cluster (symbol "running-instances")) cluster-name)
        cluster (:seon.turn.loop/cluster instance)
        handle ((requiring-resolve 'seon.cluster.agent/acquire-context!)
                (assoc cluster :seon.store/store (:seon.store/store instance)) nil {:seon.agent/branch branch})]
    (fn [source]
      ((requiring-resolve 'seon.sci.eval/evaluate)
       {:seon.cluster.eval/source source :seon.cluster.eval/ns [:seon.ns/name 'user]
        :seon.sci.eval/ctx (:seon.sci.eval/ctx handle) :seon.sci.admit/caps (:seon.sci.admit/caps cluster)
        :seon.sci.eval/time-limit-ms 30000 :seon.config/on-core-error (:seon.config/on-core-error cluster)}))))

(defn red-on-base
  "Evaluate each done-when form of each specified task; a proof that already holds proves nothing."
  [evaluate derived]
  (vec (for [t derived :when (seq (:seon.task/done-when t))
             [i form] (map-indexed vector (:seon.task/done-when t))
             :let [r (evaluate (pr-str form))
                   v (:seon.sci.admit/value r)
                   err (:seon.cluster.eval/error r)]]
         {:task (:seon.task/id t) :proof i
          :on-base (cond err :red-error (true? v) :GREEN-ALREADY :else :red)
          :detail (or (some-> err (subs 0 (min 120 (count err)))) (pr-str v))})))

(defn stale-doc-tasks
  "Detector producer: one task per namespace whose functions changed after their docstrings.
  Identity is producer + subject, so re-running finds the same tasks instead of duplicating them."
  [db]
  (let [rows (d/q '[:find ?s :where [?e :seon.fn/sym ?s] [?e :seon.fn/source _ ?tsrc] [?e :seon.fn/doc _ ?tdoc] [(> ?tsrc ?tdoc)]] db)]
    (vec (for [[ns-name syms] (group-by #(symbol (namespace %)) (map first rows))]
           {:seon.task/id (keyword "seon.task.stale-docs" (str ns-name))
            :seon.task/intent (str (count syms) " function(s) in " ns-name " changed after their docstrings; confirm each still describes the code or rewrite it.")
            :seon.task/subjects (set syms)
            :seon.task/done-when [(list 'empty? (list 'seon.task/stale-docs '(seon.db/db) (list 'quote ns-name)))]}))))

(defn attempt
  "Evaluate one task's proofs on a branch: the ATTEMPT is data, separate from the task."
  [evaluate branch task]
  (let [results (vec (for [form (:seon.task/done-when task)
                           :let [r (evaluate (pr-str form))]]
                       (cond (:seon.cluster.eval/error r) [:red (:seon.cluster.eval/error r)]
                             (true? (:seon.sci.admit/value r)) [:green]
                             :else [:red (pr-str (:seon.sci.admit/value r))])))]
    {:seon.task.attempt/task (:seon.task/id task)
     :seon.task.attempt/branch branch
     :seon.task.attempt/results results
     :seon.task.attempt/green? (boolean (and (seq results) (every? #(= :green (first %)) results)))}))

(defn states
  "Derived from tasks + attempts: settled (an attempt with every proof green), ready (specified,
  every need settled), blocked (a need unsettled), raised."
  [derived attempts]
  ;; attempts are ordered oldest → newest; only the latest attempt of a task counts (no stale green)
  (let [latest (reduce (fn [m a] (assoc m (:seon.task.attempt/task a) a)) {} attempts)
        settled (set (keep (fn [[id a]] (when (:seon.task.attempt/green? a) id)) latest))]
    (into (sorted-map)
          (for [t derived :let [id (:seon.task/id t)]]
            [id (cond (settled id) :settled
                      (= :raised (:seon.task/state t)) :raised
                      (every? settled (:seon.task/needs t)) :ready
                      :else :blocked)]))))

(defn brief
  "A v0.1 brief: the task, its proofs, its must-reads read from program rows (all real forms)."
  [db task]
  (let [row (fn [s] (d/pull db '[:seon.fn/doc :seon.fn/spec :seon.fn/source] [:seon.fn/sym s]))
        form (fn [s] (list 'seon.db/pull '(seon.db/db) ''[:seon.fn/doc :seon.fn/spec :seon.fn/source] [:seon.fn/sym (list 'quote s)]))]
    (str/join "\n\n"
              (concat
               [(str ";; Your task.\n" (with-out-str (clojure.pprint/pprint (select-keys task [:seon.task/id :seon.task/intent :seon.task/delivers]))))
                (str ";; Done when every form here returns true on your branch, each in its own evaluation.\n"
                     (with-out-str (clojure.pprint/pprint (:seon.task/done-when task))))]
               (for [[i s] (map-indexed vector (sort (:seon/must-read task))) :let [r (row s)]]
                 (str ";; Must read: " s "\nr" (+ 3 i) " => " (pr-str (form s)) "\n;; doc: " (:seon.fn/doc r)
                      "\n;; contract: " (:seon.fn/spec r) "\n" (:seon.fn/source r)))))))

(defn- literal? [x] (not (some #(or (seq? %) (symbol? %)) (tree-seq coll? seq x))))

(defn example-findings
  "Literal arguments in proof calls that fail the called function's declared input (single-arity
  :=> contracts), checked against the registry plus the task's own target schemas."
  [db registry task]
  (let [reg (mr/composite-registry (or (:seon.task/schemas task) {}) registry)]
    (vec (for [call (mapcat #(filter (fn [x] (and (seq? x) (qualified-symbol? (first x)))) (tree-seq coll? seq %))
                            (:seon.task/done-when task))
               :let [f (first call) args (vec (rest call))
                     spec (some-> (d/q '[:find ?s . :in $ ?f :where [?e :seon.fn/sym ?f] [?e :seon.fn/spec ?s]] db f)
                                  e/parse-string)]
               :when (and spec (= :=> (first spec)) (vector? (second spec)) (= :cat (first (second spec))))
               :let [inputs (vec (rest (second spec)))]
               [i arg] (map-indexed vector args)
               :when (and (< i (count inputs)) (literal? arg))
               :let [ex (try (m/explain (nth inputs i) arg {:registry reg}) (catch Exception ex {:error (ex-message ex)}))]
               :when ex]
           {:task (:seon.task/id task) :rule :example-violates-contract :fn f :argument i
            :why (if (:error ex) (:error ex) (me/humanize ex))}))))
