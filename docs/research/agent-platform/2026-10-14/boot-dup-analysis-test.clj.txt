(ns seon.boot-source-cost-test
  (:require [clojure.java.io :as io]
            [clojure.test :refer [deftest is]]
            [seon.fn :as functions]
            [seon.profile :as profile]
            [seon.test-support :as support]))

(deftest ^{:seon/tasks #{'seon.cluster/publication-tests-cost-35-seconds}}
  publication-analysis-retains-its-first-file-results
  (let [root (str "tmp/boot-source-cost-" (random-uuid))
        request {:seon.fn/root root :seon.fn/roots ["src"]
                 :seon.fn.analyzer/cache-root (str root "/resolver")}
        sources {"src/alpha.clj" "(ns boot.cost.alpha) (defn f {:malli/schema [:=> [:cat :int] :int]} [x] x)"
                 "src/beta.clj" "(ns boot.cost.beta (:require [boot.cost.alpha :as a])) (defn g {:malli/schema [:=> [:cat] :int]} [] (a/f 1))"}]
    (try
      (doseq [[path source] sources]
        (let [file (io/file root path)]
          (io/make-parents file)
          (spit file source)))
      (let [held (:seon.fn.manifest/artifacts
                  (functions/build-manifest (assoc request :seon.fn/changed-paths #{"src/alpha.clj"})))
            complete (assoc request :seon.fn/changed-paths (set (keys sources)))
            measured (profile/measure
                       #(functions/analyze-entities (assoc complete :seon.fn.file/artifacts held)))
            together (functions/analyze-entities complete)]
        (is (= (set together) (set (:seon.profile/value measured))))
        (is (= 1 (profile/calls-of measured 'seon.fn.analyzer/parsed-facts))))
      (finally (support/delete-recursively! root)))))

(deftest ^{:seon/tasks #{'seon.cluster/publication-tests-cost-35-seconds}}
  publication-continuation-costs-less-than-reanalysis
  (let [paths (into [] (take 20) (sort (filter #(re-find #"\.clj$" %)
                                               (map #(str "src/seon/" (.getName ^java.io.File %))
                                                    (.listFiles (io/file "src/seon"))))))
        request {:seon.fn/root "." :seon.fn/roots ["src"]
                 :seon.fn/changed-paths (set paths)}
        held (:seon.fn.manifest/artifacts
              (functions/build-manifest (assoc request :seon.fn/changed-paths (set (take 10 paths)))))
        change (profile/measure
                 #(functions/analyze-entities (assoc request :seon.fn.file/artifacts held)))
        parent (profile/measure #(functions/analyze-entities request))]
    (is (= (set (:seon.profile/value parent)) (set (:seon.profile/value change))))
    (is (< (:seon.profile/ms change) (:seon.profile/ms parent)))))
